    intro p q h1 h2
    have := cross_dot_cross p q p q
    rw [h2, h1, dotProduct_comm q p, h1] at this
    simp at this
    rcases this with h | h

          have := congrFun this j

          linarith
        rw [h1, hgk, h3]

        rw [h2, h1] at this

  have P : ∀ p x y : Fin 3 → ℝ, p ⨯₃ x = 0 → p ⨯₃ y = 0 → x ⨯₃ y ≠ 0 → p = 0 := by
    intro p x y h1 h2 h3
    have e1 := cross_cross p x y
    rw [h1, h2] at e1

    have e2 : p ⬝ᵥ (x ⨯₃ y) = 0 := by
