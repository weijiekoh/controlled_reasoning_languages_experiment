  constructor
  · refine ⟨convex_univ, ?_⟩
    intro a _ b _ s u hs hu hsu
    simp only [smul_eq_mul, Pi.add_apply, Pi.smul_apply, Finset.mul_sum, ← Finset.sum_add_distrib]
    apply Finset.sum_le_sum
    intro j _
    have h1 : max (ttar j - (s * a j + u * b j)) 0 ≤ s * max (ttar j - a j) 0 + u * max (ttar j - b j) 0 := by
      apply max_le
      · have := le_max_left (ttar j - a j) 0
        have := le_max_left (ttar j - b j) 0
        nlinarith [mul_le_mul_of_nonneg_left ‹ttar j - a j ≤ _› hs, mul_le_mul_of_nonneg_left ‹ttar j - b j ≤ _› hu]
      · have := le_max_right (ttar j - a j) 0
        have := le_max_right (ttar j - b j) 0
        nlinarith [mul_nonneg hs (le_max_right (ttar j - a j) 0), mul_nonneg hu (le_max_right (ttar j - b j) 0)]
    have := mul_le_mul_of_nonneg_left h1 (hw j).le
    nlinarith
  · intro a ha b hb s u hs hu hsu
    obtain ⟨a1, a2, a3, a4⟩ := ha
    obtain ⟨b1, b2, b3, b4⟩ := hb
    refine ⟨?_, ?_, ?_, ?_⟩
    · intro j
      simp only [Prod.fst_add, Prod.smul_fst, Pi.add_apply, Pi.smul_apply, smul_eq_mul]
      nlinarith [mul_nonneg hs (a1 j), mul_nonneg hu (b1 j)]
    · intro p i
      simp only [Prod.snd_add, Prod.smul_snd, Pi.add_apply, Pi.smul_apply, smul_eq_mul]
      nlinarith [mul_nonneg hs (a2 p i), mul_nonneg hu (b2 p i)]
    · intro j
      have key : ∀ p, psi p ((s • a + u • b).2 p) (∑ k, R p k * (s • a + u • b).1 k) * R p j
          ≤ s * (psi p (a.2 p) (∑ k, R p k * a.1 k) * R p j) + u * (psi p (b.2 p) (∑ k, R p k * b.1 k) * R p j) := by
        intro p
        have hc := (hpsi_convex p).2 (Set.mem_univ (a.2 p, ∑ k, R p k * a.1 k)) (Set.mem_univ (b.2 p, ∑ k, R p k * b.1 k)) hs hu hsu
        have e : ∑ k, R p k * (s • a + u • b).1 k = s * ∑ k, R p k * a.1 k + u * ∑ k, R p k * b.1 k := by
          simp only [Prod.fst_add, Prod.smul_fst, Pi.add_apply, Pi.smul_apply, smul_eq_mul, Finset.mul_sum, ← Finset.sum_add_distrib]
          apply Finset.sum_congr rfl; intro k _; ring
        rw [e]
        have hc' : psi p ((s • a + u • b).2 p) (s * ∑ k, R p k * a.1 k + u * ∑ k, R p k * b.1 k) ≤ s * psi p (a.2 p) (∑ k, R p k * a.1 k) + u * psi p (b.2 p) (∑ k, R p k * b.1 k) := by
          simpa [Prod.smul_mk, Prod.mk_add_mk] using hc
        have := mul_le_mul_of_nonneg_right hc' (hR_nonneg p j)
        nlinarith
      calc _ ≤ ∑ p, (s * (psi p (a.2 p) (∑ k, R p k * a.1 k) * R p j) + u * (psi p (b.2 p) (∑ k, R p k * b.1 k) * R p j)) := Finset.sum_le_sum (fun p _ => key p)
        _ = s * ∑ p, psi p (a.2 p) (∑ k, R p k * a.1 k) * R p j + u * ∑ p, psi p (b.2 p) (∑ k, R p k * b.1 k) * R p j := by
          rw [Finset.sum_add_distrib, Finset.mul_sum, Finset.mul_sum]
        _ ≤ s * lmax j + u * lmax j := by
          have := a3 j; have := b3 j
          nlinarith [mul_le_mul_of_nonneg_left (a3 j) hs, mul_le_mul_of_nonneg_left (b3 j) hu]
        _ = lmax j := by rw [← add_mul, hsu, one_mul]
    · intro i
      simp only [Prod.snd_add, Prod.smul_snd, Pi.add_apply, Pi.smul_apply, smul_eq_mul, Finset.sum_add_distrib, ← Finset.mul_sum]
      nlinarith [mul_le_mul_of_nonneg_left (a4 i) hs, mul_le_mul_of_nonneg_left (b4 i) hu]
