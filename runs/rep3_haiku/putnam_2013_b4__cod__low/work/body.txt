by
  have hi : ∀ k : C(Icc (0 : ℝ) 1, ℝ), Integrable (fun x => k x) := fun k =>
    k.continuous.integrable_of_hasCompactSupport (HasCompactSupport.of_compactSpace _)
  have hi' : ∀ k : Icc (0 : ℝ) 1 → ℝ, Continuous k → Integrable k := fun k hk =>
    hk.integrable_of_hasCompactSupport (HasCompactSupport.of_compactSpace _)
  -- variance is minimal among shifts
  have L : ∀ (h : C(Icc (0 : ℝ) 1, ℝ)) (c : ℝ), Var h ≤ ∫ x, (h x - c) ^ 2 := by
    intro h c
    rw [hVar h]
    have hm : ∫ x, (h x - μ h) = 0 := by
      rw [integral_sub (hi h) (integrable_const _), integral_const, ← hμ h]
      simp
    have key : ∀ x, (h x - c) ^ 2 = (h x - μ h) ^ 2 + 2 * (μ h - c) * (h x - μ h) + (μ h - c) ^ 2 := by
      intro x; ring
    have hI1 : Integrable (fun x => (h x - μ h) ^ 2) := hi' _ (by fun_prop)
    have hI2 : Integrable (fun x => 2 * (μ h - c) * (h x - μ h)) :=
      (hi' _ (by fun_prop)).const_mul _
    have hI3 : Integrable (fun _ : Icc (0 : ℝ) 1 => (μ h - c) ^ 2) := integrable_const _
    simp_rw [key]
    rw [integral_add (hI1.add hI2) hI3, integral_add hI1 hI2, integral_const_mul, hm]
    simp
    positivity
  -- bounds on sup norms
  have hfx : ∀ x, |f x| ≤ M f := fun x => (hM f).2 ⟨x, rfl⟩
  have hgx : ∀ x, |g x| ≤ M g := fun x => (hM g).2 ⟨x, rfl⟩
  have hg2 : ∀ x, (g x) ^ 2 ≤ (M g) ^ 2 := fun x => by
    have := hgx x
    nlinarith [abs_nonneg (g x), sq_abs (g x)]
  have hMf0 : 0 ≤ M f := le_trans (abs_nonneg _) (hfx ⟨0, by simp⟩)
  have ha1 : μ f ≤ M f := by
    rw [hμ f]
    have := integral_mono (hi f) (integrable_const (M f)) (fun x => (abs_le.mp (hfx x)).2)
    simpa using this
  have ha2 : -M f ≤ μ f := by
    rw [hμ f]
    have := integral_mono (integrable_const (-M f)) (hi f) (fun x => (abs_le.mp (hfx x)).1)
    simpa using this
  have ha : (μ f) ^ 2 ≤ (M f) ^ 2 := by nlinarith
  -- pointwise decomposition
  have hdec : ∀ x, (f x * g x - μ f * μ g) ^ 2
      ≤ 2 * ((f x - μ f) * g x) ^ 2 + 2 * (μ f * (g x - μ g)) ^ 2 := fun x => by
    nlinarith [sq_nonneg (((f x - μ f) * g x) - (μ f * (g x - μ g)))]
  have hIa : Integrable (fun x => ((f x - μ f) * g x) ^ 2) := hi' _ (by fun_prop)
  have hIb : Integrable (fun x => (μ f * (g x - μ g)) ^ 2) := hi' _ (by fun_prop)
  have hIc : Integrable (fun x => (g x - μ g) ^ 2) := hi' _ (by fun_prop)
  have hId : Integrable (fun x => (f x - μ f) ^ 2) := hi' _ (by fun_prop)
  have hIe : Integrable (fun x => (M g) ^ 2 * (f x - μ f) ^ 2) := hId.const_mul _
  have hIf : Integrable (fun x => (μ f) ^ 2 * (g x - μ g) ^ 2) := hIc.const_mul _
  have e1 : ∫ x, ((f x - μ f) * g x) ^ 2 ≤ ∫ x, (M g) ^ 2 * (f x - μ f) ^ 2 := by
    apply integral_mono hIa hIe
    intro x
    nlinarith [mul_nonneg (sq_nonneg (f x - μ f)) (sub_nonneg.2 (hg2 x))]
  have e2 : ∫ x, (μ f * (g x - μ g)) ^ 2 ≤ ∫ x, (M f) ^ 2 * (g x - μ g) ^ 2 := by
    apply integral_mono hIb ((hIc.const_mul _))
    intro x
    have := mul_le_mul_of_nonneg_right ha (sq_nonneg (g x - μ g))
    nlinarith
  have hsq : Var (f * g) ≤ ∫ x, (f x * g x - μ f * μ g) ^ 2 := by
    have := L (f * g) (μ f * μ g)
    simpa using this
  have hsplit : ∫ x, (f x * g x - μ f * μ g) ^ 2
      ≤ ∫ x, (2 * ((f x - μ f) * g x) ^ 2 + 2 * (μ f * (g x - μ g)) ^ 2) :=
    integral_mono ((hi' _ (by fun_prop))) (hIa.const_mul 2 |>.add (hIb.const_mul 2)) hdec
  have hexp : ∫ x, (2 * ((f x - μ f) * g x) ^ 2 + 2 * (μ f * (g x - μ g)) ^ 2)
      = 2 * ∫ x, ((f x - μ f) * g x) ^ 2 + 2 * ∫ x, (μ f * (g x - μ g)) ^ 2 := by
    rw [integral_add (hIa.const_mul 2) (hIb.const_mul 2), integral_const_mul, integral_const_mul]
  have hVf : Var f = ∫ x, (f x - μ f) ^ 2 := hVar f
  have hVg : Var g = ∫ x, (g x - μ g) ^ 2 := hVar g
  have hnn : 0 ≤ ∫ x, (g x - μ g) ^ 2 := integral_nonneg (fun x => sq_nonneg _)
  have hnf : 0 ≤ ∫ x, (f x - μ f) ^ 2 := integral_nonneg (fun x => sq_nonneg _)
  have hb : ∫ x, (μ f * (g x - μ g)) ^ 2 = (μ f) ^ 2 * ∫ x, (g x - μ g) ^ 2 := by
    rw [← integral_const_mul]
    congr 1; ext x; ring
  have hMg0 : 0 ≤ (M g) ^ 2 := sq_nonneg _
  have hc1 : ∫ x, ((f x - μ f) * g x) ^ 2 ≤ (M g) ^ 2 * Var f := by
    rw [hVf]; linarith [e1, (by rw [← integral_const_mul] : ∫ x, (M g) ^ 2 * (f x - μ f) ^ 2 = (M g) ^ 2 * ∫ x, (f x - μ f) ^ 2)]
  have hc2 : ∫ x, (μ f * (g x - μ g)) ^ 2 ≤ (M f) ^ 2 * Var g := by
    rw [hVg]; linarith [e2, (by rw [← integral_const_mul] : ∫ x, (M f) ^ 2 * (g x - μ g) ^ 2 = (M f) ^ 2 * ∫ x, (g x - μ g) ^ 2)]
  have hb2 : (μ f) ^ 2 * Var g ≤ (M f) ^ 2 * Var g := mul_le_mul_of_nonneg_right ha (by rw [hVg]; exact hnn)
  rw [hexp] at hsplit
  nlinarith [hc1, hc2, hsq, hsplit, hb2]
