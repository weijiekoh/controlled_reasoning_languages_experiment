  have : Fact p.Prime := ⟨pprime⟩
  have : NeZero p := ⟨by have := pprime.two_le; omega⟩
  set m := (p - 1) / 2 with hm
  have hp2 : 2 ≤ p := pprime.two_le
  obtain ⟨j, hj⟩ := podd
  have hpm2 : p = 2 * m + 1 := by omega
  have hm1 : 1 ≤ m := by omega
  have hmp : m < p := by omega
  have hak : ∀ k ∈ Finset.range p, a k = (k : ZMod p) ^ m := by
    intro k hk
    rw [Finset.mem_range] at hk
    rcases Nat.eq_zero_or_pos k with rfl | hkpos
    · rw [ha0.1, Nat.cast_zero, zero_pow (by omega)]
    · have h := haother ⟨k, Set.mem_Icc.mpr ⟨hkpos, by omega⟩⟩
      simp only at h
      rw [h, ZMod.natCast_mod, Nat.cast_pow]
  have hqsum : q = ∑ k ∈ Finset.range p, Polynomial.C (a k) * Polynomial.X ^ k := by
    ext n
    rw [hq n, Polynomial.finsetSum_coeff]
    simp only [Polynomial.coeff_C_mul_X_pow]
    by_cases hn : n < p
    · simp [Finset.sum_ite_eq, Finset.mem_range, hn]
    · rw [ha0.2 n (by omega)]
      simp [Finset.sum_ite_eq, Finset.mem_range, hn]
  have hT : ∀ d, (Polynomial.taylor 1 q).coeff d
      = ∑ k ∈ Finset.range p, (k : ZMod p) ^ m * ((k.choose d : ℕ) : ZMod p) := by
    intro d
    rw [hqsum, map_sum, Polynomial.finsetSum_coeff]
    apply Finset.sum_congr rfl
    intro k hk
    rw [Polynomial.taylor_mul, Polynomial.taylor_C, Polynomial.taylor_pow, Polynomial.taylor_X,
      Polynomial.C_1, Polynomial.coeff_C_mul, Polynomial.coeff_X_add_one_pow, hak k hk]
  have hsumpm : ∑ x : ZMod p, x ^ (p - 1) = -1 := by
    have h : ∀ x : ZMod p, x ^ (p - 1) = 1 - (if x = 0 then 1 else 0) := by
      intro x
      by_cases hx : x = 0
      · subst hx; simp [zero_pow (by omega : p - 1 ≠ 0)]
      · simp [hx, ZMod.pow_card_sub_one_eq_one hx]
    simp_rw [h]
    rw [Finset.sum_sub_distrib]
    simp [ZMod.card]
  have hpow : ∀ g : Polynomial (ZMod p), g.natDegree ≤ m → ∑ x : ZMod p, x ^ m * g.eval x = -(g.coeff m) := by
    intro g hg
    simp_rw [Polynomial.eval_eq_sum_range' (show g.natDegree < m + 1 by omega), Finset.mul_sum]
    rw [Finset.sum_comm]
    rw [Finset.sum_eq_single m]
    · have h1 : ∀ x : ZMod p, x ^ m * (g.coeff m * x ^ m) = g.coeff m * x ^ (p - 1) := by
        intro x
        rw [show p - 1 = m + m by omega, pow_add]; ring
      simp_rw [h1]
      rw [← Finset.mul_sum, hsumpm]; ring
    · intro i hi hne
      have hi' : i < m := by
        rw [Finset.mem_range] at hi; omega
      have h1 : ∀ x : ZMod p, x ^ m * (g.coeff i * x ^ i) = g.coeff i * x ^ (m + i) := by
        intro x; ring
      simp_rw [h1]
      rw [← Finset.mul_sum, FiniteField.sum_pow_lt_card_sub_one (ZMod p) (m + i) (by rw [ZMod.card]; omega)]
      ring
    · intro h
      exact absurd (Finset.mem_range.mpr (Nat.lt_succ_self m)) h
  have hconv : ∀ f : ZMod p → ZMod p,
      ∑ k ∈ Finset.range p, f (k : ZMod p) = ∑ x : ZMod p, f x := by
    intro f
    apply Finset.sum_nbij (fun k : ℕ => (k : ZMod p))
    · intro _ _; exact Finset.mem_univ _
    · intro a ha b hb h
      simp only [Finset.coe_range, Set.mem_Iio] at ha hb
      have := congrArg ZMod.val h
      simp only [ZMod.val_natCast] at this
      rwa [Nat.mod_eq_of_lt ha, Nat.mod_eq_of_lt hb] at this
    · intro x _
      exact ⟨x.val, Finset.mem_coe.mpr (Finset.mem_range.mpr (ZMod.val_lt x)), ZMod.natCast_zmod_val x⟩
    · intro a _; rfl
  have hD : ∀ d ≤ m, ∑ k ∈ Finset.range p, (k : ZMod p) ^ m * ((k.choose d : ℕ) : ZMod p) * (d.factorial : ZMod p)
      = -((descPochhammer (ZMod p) d).coeff m) := by
    intro d hd
    have h1 : ∀ k : ℕ, ((k.choose d : ℕ) : ZMod p) * (d.factorial : ZMod p)
        = (descPochhammer (ZMod p) d).eval (k : ZMod p) := by
      intro k
      rw [descPochhammer_eval_eq_descFactorial, Nat.descFactorial_eq_factorial_mul_choose]
      push_cast; ring
    simp_rw [mul_assoc, h1]
    have h2 := hconv (fun x => x ^ m * (descPochhammer (ZMod p) d).eval x)
    rw [h2, hpow _ (by rw [descPochhammer_natDegree]; exact hd)]
  have hcoefD : ∀ d ≤ m, (descPochhammer (ZMod p) d).coeff m = if d = m then 1 else 0 := by
    intro d hd
    rcases Nat.lt_or_ge d m with h | h
    · rw [if_neg (by omega)]
      apply Polynomial.coeff_eq_zero_of_natDegree_lt
      rw [descPochhammer_natDegree]; exact h
    · have hdm : d = m := by omega
      have := (monic_descPochhammer (ZMod p) d).coeff_natDegree
      rw [descPochhammer_natDegree] at this
      rw [if_pos hdm, ← hdm, this]
  have key : ∀ d ≤ m, (∑ k ∈ Finset.range p, (k : ZMod p) ^ m * ((k.choose d : ℕ) : ZMod p))
      * (d.factorial : ZMod p) = -(if d = m then 1 else 0) := by
    intro d hd
    rw [Finset.sum_mul, hD d hd, hcoefD d hd]
  have hfact : ∀ d < p, (d.factorial : ZMod p) ≠ 0 := by
    intro d hd hz
    rw [ZMod.natCast_eq_zero_iff] at hz
    have := (Nat.Prime.dvd_factorial pprime).mp hz
    omega
  have hzero : ∀ d < m, (Polynomial.taylor 1 q).coeff d = 0 := by
    intro d hd
    rw [hT d]
    have h := key d (by omega)
    rw [if_neg (by omega), neg_zero] at h
    exact (mul_eq_zero.mp h).resolve_right (hfact d (by omega))
  have hnz : (Polynomial.taylor 1 q).coeff m ≠ 0 := by
    rw [hT m]
    intro hS
    have h := key m le_rfl
    rw [if_pos rfl, hS, zero_mul] at h
    exact one_ne_zero (neg_eq_zero.mp h.symm)
  have L1 : ∀ n, (Polynomial.X - 1) ^ n ∣ q → Polynomial.X ^ n ∣ Polynomial.taylor 1 q := by
    rintro n ⟨r, hr⟩
    refine ⟨Polynomial.taylor 1 r, ?_⟩
    rw [hr, Polynomial.taylor_mul, Polynomial.taylor_pow]
    congr 1
    simp
  have L2 : ∀ n, Polynomial.X ^ n ∣ Polynomial.taylor 1 q → (Polynomial.X - 1) ^ n ∣ q := by
    rintro n ⟨r, hr⟩
    refine ⟨Polynomial.taylor (-1) r, ?_⟩
    have hq' : q = Polynomial.taylor (-1) (Polynomial.taylor 1 q) := by
      rw [Polynomial.taylor_taylor]; simp
    rw [hq', hr, Polynomial.taylor_mul, Polynomial.taylor_pow, Polynomial.taylor_X]
    congr 1
    simp [sub_eq_add_neg]
  constructor
  · show npoly m ∣ q
    rw [hnpoly]
    exact L2 m (Polynomial.X_pow_dvd_iff.mpr (fun d hd => hzero d hd))
  · intro n hn
    show n ≤ m
    rw [Set.mem_setOf_eq, hnpoly] at hn
    by_contra hlt
    have h1 : (Polynomial.X - 1) ^ (m + 1) ∣ q :=
      (pow_dvd_pow (Polynomial.X - 1) (by omega : m + 1 ≤ n)).trans hn
    have h2 := Polynomial.X_pow_dvd_iff.mp (L1 (m + 1) h1) m (by omega)
    exact hnz h2
