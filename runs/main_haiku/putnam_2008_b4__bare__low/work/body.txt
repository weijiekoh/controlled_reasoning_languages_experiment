  have cong : ∀ (g : Polynomial ℤ) (m a b : ℤ), a ≡ b [ZMOD m] → g.eval a ≡ g.eval b [ZMOD m] := by
    intro g m a b hab
    rw [Int.modEq_iff_dvd] at hab ⊢
    exact dvd_trans hab (Polynomial.sub_dvd_eval_sub b a g)
  have rep : ∀ (m a : ℤ), 0 < m → ∃ r : ℕ, (r:ℤ) < m ∧ (r:ℤ) ≡ a [ZMOD m] := by
    intro m a hm
    have h0 := Int.emod_nonneg a hm.ne'
    have h1 := Int.emod_lt_of_pos a hm
    refine ⟨(a % m).toNat, ?_, ?_⟩
    · rw [Int.toNat_of_nonneg h0]; exact h1
    · rw [Int.toNat_of_nonneg h0]
      exact Int.mod_modEq a m
  have hp2 : (2:ℤ) ≤ p := by exact_mod_cast hp.two_le
  have hpz : (0:ℤ) < p := by linarith
  have hD : ∀ a : ℤ, ¬ (p:ℤ) ∣ (Polynomial.derivative h).eval a := by
    intro a ha
    obtain ⟨r, hr, hra⟩ := rep ((p:ℤ)^2) a (by positivity)
    have hrp : (r:ℤ) ≡ a [ZMOD p] := Int.ModEq.of_dvd (dvd_pow_self (p:ℤ) two_ne_zero) hra

      have h2 := (cong (Polynomial.derivative h) p r a hrp).trans (Int.modEq_zero_iff_dvd.mpr ha)
      exact Int.modEq_zero_iff_dvd.mp h2
    obtain ⟨s, hs, hsr⟩ := rep ((p:ℤ)^2) ((r:ℤ) + p) (by positivity)
    obtain ⟨K, hK⟩ := h.binomExpansion (r:ℤ) (p:ℤ)
    obtain ⟨c, hc⟩ := hDr
    have e1 : h.eval (r:ℤ) ≡ h.eval ((r:ℤ)+p) [ZMOD (p:ℤ)^2] := by
      rw [hK, hc]
      apply Int.modEq_iff_dvd.mpr
      exact ⟨c + K, by ring⟩
    have h1 : h.eval (r:ℤ) ≡ h.eval (s:ℤ) [ZMOD (p:ℤ)^2] :=
      e1.trans (cong h _ _ _ hsr.symm)
    have hr' : r < p^2 := by exact_mod_cast hr
    have hs' : s < p^2 := by exact_mod_cast hs
    have hrs : r = s := by
      have := hh ⟨r, Finset.mem_range.mpr hr'⟩ ⟨s, Finset.mem_range.mpr hs'⟩ h1
      simpa using this
    subst hrs
    rw [Int.modEq_iff_dvd] at hsr
    have h3 : (p:ℤ)^2 ∣ p := by simpa using hsr
    have := Int.le_of_dvd hpz h3
    nlinarith
  intro i j hij
  have hmod2 : h.eval (i.1:ℤ) ≡ h.eval (j.1:ℤ) [ZMOD (p:ℤ)^2] :=
    Int.ModEq.of_dvd (pow_dvd_pow (p:ℤ) (by norm_num : 2 ≤ 3)) hij
  have hdiv : (p:ℤ)^2 ∣ (j.1:ℤ) - i.1 := by

    obtain ⟨s, hs, hsb⟩ := rep ((p:ℤ)^2) j.1 (by positivity)

    have hs' : s < p^2 := by exact_mod_cast hs

      have := hh ⟨r, Finset.mem_range.mpr hr'⟩ ⟨s, Finset.mem_range.mpr hs'⟩ hrs
      simpa using this
    subst hEq
    exact Int.modEq_iff_dvd.mp (hra.symm.trans hsb)
  obtain ⟨c, hc⟩ := hdiv
  obtain ⟨K, hK⟩ := h.binomExpansion (i.1:ℤ) ((j.1:ℤ) - i.1)
  have hK' : h.eval (j.1:ℤ) = h.eval (i.1:ℤ) + (Polynomial.derivative h).eval (i.1:ℤ) * ((j.1:ℤ) - i.1) + K * ((j.1:ℤ) - i.1)^2 := by
    have := hK
    rwa [show (i.1:ℤ) + ((j.1:ℤ) - i.1) = j.1 by ring] at this
  rw [hc] at hK'
  have hij' : (p:ℤ)^3 ∣ h.eval (j.1:ℤ) - h.eval (i.1:ℤ) := Int.modEq_iff_dvd.mp hij
  have e : (Polynomial.derivative h).eval (i.1:ℤ) * ((p:ℤ)^2 * c)
      = (h.eval (j.1:ℤ) - h.eval (i.1:ℤ)) - (p:ℤ)^3 * (K * p * c^2) := by
    linear_combination -hK'

    convert h4 using 1 <;> ring

    · exact absurd h7 (hD _)
    · exact h7

  have hj0 : (0:ℤ) ≤ j.1 := by positivity

    by_contra hcon; push_neg at hcon

    by_contra hcon; push_neg at hcon
    nlinarith [mul_le_mul_of_nonneg_left hcon hP.le]

    rw [hm0] at hjm; linarith
